cmake_minimum_required(VERSION 3.10)

set(METABUF_LIBRARY_NAME metabuf)

PROJECT(metabuf)

OPTION(METABUF_EXTERNAL_BUILD "METABUF_EXTERNAL_BUILD" OFF)
OPTION(METABUF_METAGEN "METABUF_METAGEN" ON)
OPTION(METABUF_XML2METABUF "METABUF_XML2METABUF" OFF)
OPTION(METABUF_JSON2METABUF "METABUF_JSON2METABUF" OFF)
OPTION(METABUF_CONVERTER "METABUF_CONVERTER" OFF)
OPTION(METABUF_PUGIXML_EXTERNAL "METABUF_PUGIXML_EXTERNAL" OFF)
OPTION(METABUF_JANSSON_EXTERNAL "METABUF_JANSSON_EXTERNAL" OFF)
OPTION(METABUF_EXTERNAL_CONFIG "METABUF_EXTERNAL_CONFIG" OFF)
OPTION(METABUF_INSTALL "METABUF_INSTALL" OFF)
OPTION(METABUF_TEST "METABUF_TEST" ON)

set(METABUF_PUGIXML_INCLUDE_DIRS "" CACHE STRING "METABUF_PUGIXML_INCLUDE_DIRS")
set(METABUF_PUGIXML_LIBRARIES "" CACHE STRING "METABUF_PUGIXML_LIBRARIES")
set(METABUF_PUGIXML_TARGET "" CACHE STRING "METABUF_PUGIXML_TARGET")
set(METABUF_JANSSON_INCLUDE_DIRS "" CACHE STRING "METABUF_JANSSON_INCLUDE_DIRS")
set(METABUF_JANSSON_LIBRARIES "" CACHE STRING "METABUF_JANSSON_LIBRARIES")
set(METABUF_JANSSON_TARGET "" CACHE STRING "METABUF_JANSSON_TARGET")

IF(NOT METABUF_EXTERNAL_BUILD)
    if(${CMAKE_C_COMPILER_ID} STREQUAL Clang)
        set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -Wextra -Werror")
    elseif(${CMAKE_C_COMPILER_ID} STREQUAL AppleClang)
        set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -Wextra -Werror")
    elseif(${CMAKE_C_COMPILER_ID} STREQUAL GNU)
        set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -Wextra -Werror")
    elseif(${CMAKE_C_COMPILER_ID} STREQUAL Intel)
        set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -Wextra -Werror")
    elseif(${CMAKE_C_COMPILER_ID} STREQUAL MSVC)
        set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} /W4 /WX")
        set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} /wd4710")
        set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} /wd4738")
        set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} /wd4820")
        set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} /wd4711")
    endif()

    set(CMAKE_CXX_STANDARD 14)
    
    if(MSVC)
        add_definitions(-D_CRT_SECURE_NO_WARNINGS)
        add_definitions(-D_SCL_SECURE_NO_WARNINGS)

        set(CMAKE_CXX_FLAGS_DEBUG "${CMAKE_CXX_FLAGS_DEBUG} /MTd")
        set(CMAKE_CXX_FLAGS_RELEASE "${CMAKE_CXX_FLAGS_RELEASE} /MT")
        set(CMAKE_CXX_FLAGS_RELWITHDEBINFO "${CMAKE_CXX_FLAGS_RELWITHDEBINFO} /MT")

        set(CMAKE_C_FLAGS_DEBUG "${CMAKE_C_FLAGS_DEBUG} /MTd")
        set(CMAKE_C_FLAGS_RELEASE "${CMAKE_C_FLAGS_RELEASE} /MT")
        set(CMAKE_C_FLAGS_RELWITHDEBINFO "${CMAKE_C_FLAGS_RELWITHDEBINFO} /MT")    
    endif()
ENDIF()

IF(METABUF_EXTERNAL_CONFIG)
    add_definitions(-DMETABUF_EXTERNAL_CONFIG)
ENDIF()

set(METABUF_INCLUDE_PATH ${CMAKE_CURRENT_SOURCE_DIR}/include)
set(METABUF_SOURCE_PATH ${CMAKE_CURRENT_SOURCE_DIR}/src)

SET(METABUF_SOURCE_FILES ${METABUF_SOURCE_FILES} ${METABUF_INCLUDE_PATH}/config/Metaconfig.hpp)
SET(METABUF_SOURCE_FILES ${METABUF_SOURCE_FILES} ${METABUF_INCLUDE_PATH}/metabuf/Metadata.hpp)
SET(METABUF_SOURCE_FILES ${METABUF_SOURCE_FILES} ${METABUF_INCLUDE_PATH}/metabuf/Reader.hpp)
SET(METABUF_SOURCE_FILES ${METABUF_SOURCE_FILES} ${METABUF_SOURCE_PATH}/metabuf/Metadata.cpp)

ADD_LIBRARY(metabuf STATIC ${METABUF_SOURCE_FILES})

TARGET_INCLUDE_DIRECTORIES(metabuf
    PUBLIC
        ${METABUF_INCLUDE_PATH}
    PRIVATE
        ${METABUF_SOURCE_PATH}/metabuf
)

if(METABUF_INSTALL)
    install(DIRECTORY include
        DESTINATION .
        FILES_MATCHING PATTERN "*.hpp" PATTERN "*.h")

    install(TARGETS metabuf
        ARCHIVE DESTINATION lib)
endif()

set(METABUF_BUILD_XML2METABUF OFF)

if(METABUF_METAGEN OR METABUF_XML2METABUF OR METABUF_JSON2METABUF OR METABUF_CONVERTER OR METABUF_TEST)
    set(METABUF_BUILD_XML2METABUF ON)

    if(NOT METABUF_PUGIXML_EXTERNAL)
        include(cmake/pugixml_download.cmake)
        
        set(METABUF_PUGIXML_INCLUDE_DIRS ${pugixml_INCLUDE_DIR})
        set(METABUF_PUGIXML_LIBRARIES ${pugixml_LIBRARY_NAME})
    endif()
endif()

if(METABUF_BUILD_XML2METABUF)
    #xml2metabuf

    set(XML2METABUF_SOURCE_FILES ${XML2METABUF_SOURCE_FILES} ${METABUF_SOURCE_PATH}/xml2metabuf/Xml2Metabuf.cpp)
    set(XML2METABUF_SOURCE_FILES ${XML2METABUF_SOURCE_FILES} ${METABUF_SOURCE_PATH}/xml2metabuf/Xml2Metabuf.hpp)
    set(XML2METABUF_SOURCE_FILES ${XML2METABUF_SOURCE_FILES} ${METABUF_SOURCE_PATH}/xml2metabuf/Xml2Metacode.cpp)
    set(XML2METABUF_SOURCE_FILES ${XML2METABUF_SOURCE_FILES} ${METABUF_SOURCE_PATH}/xml2metabuf/Xml2Metacode.hpp)
    set(XML2METABUF_SOURCE_FILES ${XML2METABUF_SOURCE_FILES} ${METABUF_SOURCE_PATH}/xml2metabuf/Xml2Metaprotocol.cpp)
    set(XML2METABUF_SOURCE_FILES ${XML2METABUF_SOURCE_FILES} ${METABUF_SOURCE_PATH}/xml2metabuf/Xml2Metaprotocol.hpp)
    set(XML2METABUF_SOURCE_FILES ${XML2METABUF_SOURCE_FILES} ${METABUF_SOURCE_PATH}/xml2metabuf/XmlProtocol.cpp)
    set(XML2METABUF_SOURCE_FILES ${XML2METABUF_SOURCE_FILES} ${METABUF_SOURCE_PATH}/xml2metabuf/XmlProtocol.hpp)
    set(XML2METABUF_SOURCE_FILES ${XML2METABUF_SOURCE_FILES} ${METABUF_SOURCE_PATH}/xml2metabuf/XmlCRC32.cpp)
    set(XML2METABUF_SOURCE_FILES ${XML2METABUF_SOURCE_FILES} ${METABUF_SOURCE_PATH}/xml2metabuf/XmlCRC32.hpp)
    set(XML2METABUF_SOURCE_FILES ${XML2METABUF_SOURCE_FILES} ${METABUF_SOURCE_PATH}/xml2metabuf/XmlConverter.cpp)
    set(XML2METABUF_SOURCE_FILES ${XML2METABUF_SOURCE_FILES} ${METABUF_SOURCE_PATH}/xml2metabuf/XmlConverter.hpp)

    ADD_LIBRARY(xml2metabuf STATIC ${XML2METABUF_SOURCE_FILES})

    TARGET_INCLUDE_DIRECTORIES(xml2metabuf
        PUBLIC
            ${METABUF_INCLUDE_PATH}
            ${METABUF_SOURCE_PATH}
            ${METABUF_PUGIXML_INCLUDE_DIRS}
    )

    TARGET_LINK_LIBRARIES(xml2metabuf
        PUBLIC
            ${METABUF_PUGIXML_LIBRARIES}
    )

    if(METABUF_PUGIXML_TARGET)
        ADD_DEPENDENCIES(xml2metabuf ${METABUF_PUGIXML_TARGET})
    elseif(NOT METABUF_PUGIXML_EXTERNAL)
        ADD_DEPENDENCIES(xml2metabuf pugixml_download)
    endif()
endif()

if(METABUF_CONVERTER)
    #metabufconverter

    set(METABUFCONVERTER_SOURCE_FILES ${METABUFCONVERTER_SOURCE_FILES} ${METABUF_SOURCE_PATH}/metabufconverter/Converter.cpp)
    set(METABUFCONVERTER_SOURCE_FILES ${METABUFCONVERTER_SOURCE_FILES} ${METABUF_SOURCE_PATH}/metabufconverter/Converter.hpp)
    set(METABUFCONVERTER_SOURCE_FILES ${METABUFCONVERTER_SOURCE_FILES} ${METABUF_SOURCE_PATH}/metabufconverter/OutputAdapter.hpp)

    ADD_LIBRARY(metabufconverter STATIC ${METABUFCONVERTER_SOURCE_FILES})

    TARGET_INCLUDE_DIRECTORIES(metabufconverter
        PUBLIC
            ${METABUF_INCLUDE_PATH}
            ${METABUF_SOURCE_PATH}
    )

    TARGET_LINK_LIBRARIES(metabufconverter
        PUBLIC
            xml2metabuf
    )
endif()

if(METABUF_JSON2METABUF)
    #json2metabuf

    if(NOT METABUF_JANSSON_EXTERNAL)
        message(FATAL_ERROR "METABUF_JSON2METABUF requires METABUF_JANSSON_EXTERNAL")
    endif()

    set(JSON2METABUF_SOURCE_FILES ${JSON2METABUF_SOURCE_FILES} ${METABUF_SOURCE_PATH}/json2metabuf/JsonConverter.cpp)
    set(JSON2METABUF_SOURCE_FILES ${JSON2METABUF_SOURCE_FILES} ${METABUF_SOURCE_PATH}/json2metabuf/JsonConverter.hpp)
    set(JSON2METABUF_SOURCE_FILES ${JSON2METABUF_SOURCE_FILES} ${METABUF_SOURCE_PATH}/json2metabuf/Json2Metabuf.cpp)
    set(JSON2METABUF_SOURCE_FILES ${JSON2METABUF_SOURCE_FILES} ${METABUF_SOURCE_PATH}/json2metabuf/Json2Metabuf.hpp)

    ADD_LIBRARY(json2metabuf STATIC ${JSON2METABUF_SOURCE_FILES})

    TARGET_INCLUDE_DIRECTORIES(json2metabuf
        PUBLIC
            ${METABUF_INCLUDE_PATH}
            ${METABUF_SOURCE_PATH}
            ${METABUF_PUGIXML_INCLUDE_DIRS}
            ${METABUF_JANSSON_INCLUDE_DIRS}
    )

    TARGET_LINK_LIBRARIES(json2metabuf
        PUBLIC
            xml2metabuf
            ${METABUF_JANSSON_LIBRARIES}
    )

    if(METABUF_JANSSON_TARGET)
        ADD_DEPENDENCIES(json2metabuf ${METABUF_JANSSON_TARGET})
    endif()
endif()

if(METABUF_METAGEN)
    #metagen

    set(METAGEN_SOURCE_FILES ${METAGEN_SOURCE_FILES} ${METABUF_SOURCE_PATH}/metagen/metagen.cpp)

    ADD_EXECUTABLE(metagen ${METAGEN_SOURCE_FILES})

    TARGET_INCLUDE_DIRECTORIES(metagen
        PRIVATE
            ${METABUF_INCLUDE_PATH}
            ${METABUF_SOURCE_PATH}
            ${METABUF_SOURCE_PATH}/xml2metabuf
            ${METABUF_PUGIXML_INCLUDE_DIRS}
    )

    if(WIN32)
        set_target_properties(metagen PROPERTIES LINK_FLAGS "/SUBSYSTEM:CONSOLE")
        set_target_properties(metagen PROPERTIES COMPILE_DEFINITIONS "_CONSOLE")
    endif()

    TARGET_LINK_LIBRARIES(metagen
        PRIVATE
            metabuf
            xml2metabuf
    )
endif()

if(METABUF_TEST)
    include(CTest)
    enable_testing()

    add_subdirectory(tests)
endif(METABUF_TEST)
